torch>=2.2
torchaudio>=2.2
# Exact-pinned rather than a lower bound: transformers 5.x renamed/removed several
# Trainer kwargs (e.g. tokenizer= -> processing_class=, verified by reading
# transformers/trainer.py's source directly for this version). An unbounded
# `>=` here previously let a fresh `pip install` silently jump major versions
# mid-project and break afrimeet/models/train.py with no warning.
transformers==5.14.1
# Pinned <3.0: our Common Voice source (fsicoli/common_voice_*, see configs/config.yaml)
# still ships a legacy Python loading script, which `datasets` >=3.0 refuses to execute
# at all (HF's Sept 2024 script-datasets policy change) -- there's currently no
# already-converted, no-script Common Voice mirror for Swahili. HF Hub still serves the
# script file itself, so the older client library still works; only the 3.0+ client
# added the client-side refusal. Revisit this pin if a parquet/audiofolder Common Voice
# Swahili mirror shows up.
datasets>=2.19,<3.0
# datasets depends on aiohttp with no floor of its own; without a constraint here pip's
# resolver picked aiohttp 1.0.5 (~2015) in one local install, which fails to import
# entirely on modern Python (`asyncio.async` was removed). Floor it explicitly.
aiohttp>=3.9
accelerate>=0.30
evaluate>=0.4
# Exact-pinned: jiwer 4.0 renamed wer()/cer()'s truth_transform kwarg to

librosa>=0.10
